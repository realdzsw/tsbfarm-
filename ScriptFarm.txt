-- ESP Player Logic
local function updateEspPlayer()
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and p.Character then
            local char = p.Character
            if espPlayerActive then
                if not char:FindFirstChild("PlayerESP_HL") then
                    local hl = Instance.new("Highlight")
                    hl.Name = "PlayerESP_HL"
                    hl.FillColor = Color3.fromRGB(0, 255, 255)
                    hl.FillTransparency = 0.5
                    hl.OutlineColor = Color3.fromRGB(255, 255, 255)
                    hl.Adornee = char
                    hl.Parent = char
                end
                local head = char:FindFirstChild("Head")
                if head and not head:FindFirstChild("PlayerESP_Name") then
                    local bg = Instance.new("BillboardGui")
                    bg.Name = "PlayerESP_Name"
                    bg.Size = UDim2.new(0, 100, 0, 40)
                    bg.StudsOffset = Vector3.new(0, 2.5, 0)
                    bg.AlwaysOnTop = true
                    bg.Parent = head

                    local txt = Instance.new("TextLabel")
                    txt.Size = UDim2.new(1, 0, 1, 0)
                    txt.BackgroundTransparency = 1
                    txt.Text = p.Name
                    txt.TextColor3 = Color3.fromRGB(255, 255, 0)
                    txt.TextStrokeTransparency = 0
                    txt.Font = Enum.Font.GothamBold
                    txt.TextSize = 12
                    txt.Parent = bg
                end
            else
                local hl = char:FindFirstChild("PlayerESP_HL")
                if hl then hl:Destroy() end
                local head = char:FindFirstChild("Head")
                if head then
                    local bg = head:FindFirstChild("PlayerESP_Name")
                    if bg then bg:Destroy() end
                end
            end
        end
    end
end

RunService.Heartbeat:Connect(function()

-- Server Hop
serverHopBtn.MouseButton1Click:Connect(function()
    status.Text = "Status: Đang tìm Server..."
    task.spawn(function()
        local placeId = game.PlaceId
        local req = request or http_request or (syn and syn.request)
        local lowestServer, minPlayers = nil, math.huge
        pcall(function()
            local res = req({Url = string.format("https://games.roblox.com/v1/games/%d/servers/Public?sortOrder=Asc&limit=100", placeId), Method = "GET"})
            if res and res.Body then
                local data = HttpService:JSONDecode(res.Body)
                if data and data.data then
                    for _, s in ipairs(data.data) do
                        if s.id ~= game.JobId and s.playing < s.maxPlayers and s.playing < minPlayers then
                            minPlayers = s.playing; lowestServer = s.id
                        end
                    end
                end
            end
        end)
        if lowestServer then TeleportService:TeleportToPlaceInstance(placeId, lowestServer, LocalPlayer)
        else status.Text = "Không tìm thấy Server!" end
    end)
end)

-- ========== AUTO FARM LOGIC ==========
local function createChams(obj)
    if not obj:FindFirstChild("HalloweenCham") then
        local hl = Instance.new("Highlight")
        hl.Name = "HalloweenCham"; hl.FillColor = CHAMS_COLOR; hl.FillTransparency = 0.4
        hl.OutlineColor = Color3.fromRGB(255, 255, 255); hl.Adornee = obj; hl.Parent = obj
    end
end

local function clearAllChams()
    for _, obj in ipairs(workspace:GetDescendants()) do
        if obj:IsA("Highlight") and obj.Name == "HalloweenCham" then obj:Destroy() end
    end
end

local function forceTweenTo(targetCFrame)
    local char = LocalPlayer.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    local hum = char and char:FindFirstChildWhichIsA("Humanoid")
    if not root or not hum then return end
    local dist = (root.Position - targetCFrame.Position).Magnitude
    local tween = TweenService:Create(root, TweenInfo.new(dist / TWEEN_SPEED, Enum.EasingStyle.Linear), {CFrame = targetCFrame})
    local conn = RunService.Heartbeat:Connect(function()
        if root and hum then root.AssemblyLinearVelocity = Vector3.new(0,0,0); hum.PlatformStand = false end
    end)
    tween:Play(); tween.Completed:Wait()
    if conn then conn:Disconnect() end
end

local function findTarget()
    local char = LocalPlayer.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    if not root then return nil, nil end
    local chosenPart, chosenPrompt, bestDist = nil, nil, math.huge
    for _, v in ipairs(workspace:GetDescendants()) do
        if v:IsA("ProximityPrompt") then
            local pObj = v.Parent
            local pName = pObj and pObj.Name:lower() or ""
            local oText, aText = v.ObjectText:lower(), v.ActionText:lower()
            local isTrash = pName:match("trash") or pName:match("bin") or pName:match("can")
            local isHW = oText:match("trick") or aText:match("treat") or pName:match("door") or pName:match("house") or pName:match("event")
            if isHW and not isTrash and not IgnoredObjects[pObj] then
                local part = pObj:IsA("BasePart") and pObj or pObj:FindFirstChildWhichIsA("BasePart")
                if part then
                    local dist = (root.Position - part.Position).Magnitude
                    if dist < bestDist then bestDist = dist; chosenPart = part; chosenPrompt = v end
                end
            end
        end
    end
    return chosenPart, chosenPrompt
end

local farmThread = nil
toggleBtn.MouseButton1Click:Connect(function()
    FARM_ENABLED = not FARM_ENABLED
    toggleBtn.Text = "Auto Farm Halloween: " .. (FARM_ENABLED and "ON" or "OFF")
    setButtonState(toggleBtn, FARM_ENABLED)
    if FARM_ENABLED then
        IgnoredObjects = {}
        if farmThread then task.cancel(farmThread) end
        farmThread = task.spawn(function()
            status.Text = "Status: Hunting..."; status.TextColor3 = THEME.SUCCESS
            while FARM_ENABLED do
                task.wait(0.05)
                local target, prompt = findTarget()
                if target and prompt then
                    createChams(target)
                    forceTweenTo(target.CFrame * CFrame.new(0, 0, INTERACT_DISTANCE))
                    status.Text = "Target: " .. tostring(target.Name)
                    task.wait(WAIT_AT_DOOR)
                    if fireproximityprompt then fireproximityprompt(prompt)
                    else prompt:InputHoldBegin(); task.wait(prompt.HoldDuration + 0.05); prompt:InputHoldEnd() end
                    IgnoredObjects[prompt.Parent] = true
                    status.Text = "Cooldown..."
                    task.wait(COOLDOWN_DURATION)
                else
                    status.Text = "Status: Searching targets..."; task.wait(1)
                end
            end
            status.Text = "Status: Stopped"; status.TextColor3 = THEME.MUTED
        end)
    else
        if farmThread then task.cancel(farmThread); farmThread = nil end
        clearAllChams()
        status.Text = "Status: Off"; status.TextColor3 = THEME.MUTED
    end
end)
