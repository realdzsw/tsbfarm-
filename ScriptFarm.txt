-- ========== SERVER HOP (Ít người nhất) ==========
local function hopLowestServer()
    status.Text = "Status: Đang tìm Server ít người..."
    serverHopBtn.Text = "Đang quét Server..."
    
    task.spawn(function()
        local placeId = game.PlaceId
        local req = request or http_request or (syn and syn.request) or (http and http.request)
        
        if not req then
            status.Text = "Status: Executor không hỗ trợ!"
            serverHopBtn.Text = "🚀 Server Hop (Ít người nhất)"
            return
        end
        
        local lowestServer = nil
        local minPlayers = math.huge
        
        pcall(function()
            local url = string.format("https://games.roblox.com/v1/games/%d/servers/Public?sortOrder=Asc&limit=100", placeId)
            local res = req({Url = url, Method = "GET"})
            if res and res.Body then
                local bodyData = HttpService:JSONDecode(res.Body)
                if bodyData and bodyData.data then
                    for _, s in ipairs(bodyData.data) do
                        if s.id ~= game.JobId and s.playing < s.maxPlayers and s.playing > 0 then
                            if s.playing < minPlayers then
                                minPlayers = s.playing
                                lowestServer = s.id
                            end
                        end
                    end
                end
            end
        end)
        
        if lowestServer then
            status.Text = string.format("Đang vào Server (%d người)...", minPlayers)
            TeleportService:TeleportToPlaceInstance(placeId, lowestServer, LocalPlayer)
        else
            status.Text = "Không tìm thấy Server!"
            serverHopBtn.Text = "🚀 Server Hop (Ít người nhất)"
        end
    end)
end

serverHopBtn.MouseButton1Click:Connect(hopLowestServer)

    frameCount = frameCount + 1
    if tick() - lastTick >= 1 then
        local fps = math.floor(frameCount / (tick() - lastTick))

        pcall(function()

        end)
        statsLabel.Text = string.format("⚡ FPS: %d | 🏓 Ping: %dms", fps, ping)
        frameCount = 0
        lastTick = tick()
    end
end)

-- ========== CHAMS & MOVEMENT LOGIC ==========
local function createChams(object)
    if not object:FindFirstChild("HalloweenCham") then
        local highlight = Instance.new("Highlight")
        highlight.Name = "HalloweenCham"
        highlight.FillColor = CHAMS_COLOR
        highlight.FillTransparency = 0.4
        highlight.OutlineColor = Color3.fromRGB(255, 255, 255)
        highlight.Adornee = object
        highlight.Parent = object
    end
end

local function clearAllChams()
    for _, obj in ipairs(workspace:GetDescendants()) do
        if obj:IsA("Highlight") and obj.Name == "HalloweenCham" then
            obj:Destroy()
        end
    end
end

local function forceTweenTo(targetCFrame)
    local character = LocalPlayer.Character
    local rootPart = character and character:FindFirstChild("HumanoidRootPart")
    local humanoid = character and character:FindFirstChildWhichIsA("Humanoid")
    if not rootPart or not humanoid then return end

    local distance = (rootPart.Position - targetCFrame.Position).Magnitude
    local duration = distance / TWEEN_SPEED

    local tweenInfo = TweenInfo.new(duration, Enum.EasingStyle.Linear)
    local tween = TweenService:Create(rootPart, tweenInfo, {CFrame = targetCFrame})
    
    local forceConnection
    forceConnection = RunService.Heartbeat:Connect(function()
        if character and rootPart and humanoid then
            rootPart.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
            rootPart.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
            humanoid.PlatformStand = false
            if humanoid:GetState() == Enum.HumanoidStateType.Ragdoll or humanoid:GetState() == Enum.HumanoidStateType.Physics then
                humanoid:ChangeState(Enum.HumanoidStateType.GettingUp)
            end
        end
    end)

    tween:Play()
    tween.Completed:Wait()
    
    if forceConnection then 
        forceConnection:Disconnect() 
    end
end

local function findTarget()
    local character = LocalPlayer.Character
    local rootPart = character and character:FindFirstChild("HumanoidRootPart")
    if not rootPart then return nil, nil end

    local chosenPart = nil
    local chosenPrompt = nil
    local bestDistance = math.huge

    for _, v in ipairs(workspace:GetDescendants()) do
        if v:IsA("ProximityPrompt") then
            local parentObj = v.Parent
            local parentName = parentObj and parentObj.Name:lower() or ""
            local objText = v.ObjectText:lower()
            local actionText = v.ActionText:lower()
            
            local isTrash = parentName:match("trash") or parentName:match("bin") or parentName:match("can")
            local isHalloween = objText:match("trick") or actionText:match("treat") or parentName:match("door") or parentName:match("house") or parentName:match("event")
            
            if isHalloween and not isTrash and not IgnoredObjects[parentObj] then
                local interactablePart = parentObj:IsA("BasePart") and parentObj or parentObj:FindFirstChildWhichIsA("BasePart")
                if interactablePart then
                    local dist = (rootPart.Position - interactablePart.Position).Magnitude
                    if dist < bestDistance then
                        bestDistance = dist
                        chosenPart = interactablePart
                        chosenPrompt = v
                    end
                end
            end
        end
    end
    return chosenPart, chosenPrompt
end

-- ========== MAIN LOOP ==========
local farmThread = nil

local function startFarm()
    if farmThread then task.cancel(farmThread) end
    farmThread = task.spawn(function()
        status.Text = "Status: Hunting..."
        status.TextColor3 = THEME.SUCCESS
        
        while FARM_ENABLED do
            task.wait(0.05)
            
            local currentTarget, currentPrompt = findTarget()
            
            if currentTarget and currentPrompt then
                createChams(currentTarget)
                
                local destination = currentTarget.CFrame * CFrame.new(0, 0, INTERACT_DISTANCE)
                status.Text = "Target: " .. tostring(currentTarget.Name)
                forceTweenTo(destination)
                
                status.Text = "Touching door..."
                task.wait(WAIT_AT_DOOR)
                
                if fireproximityprompt then
                    fireproximityprompt(currentPrompt)
                else
                    currentPrompt:InputHoldBegin()
                    task.wait(currentPrompt.HoldDuration + 0.05)
                    currentPrompt:InputHoldEnd()
                end
                
                IgnoredObjects[currentPrompt.Parent] = true
                
                status.Text = "Cooldown..."
                task.wait(COOLDOWN_DURATION)
            else
                status.Text = "Status: Searching targets..."
                task.wait(1)
            end
        end
        
        status.Text = "Status: Stopped"
        status.TextColor3 = THEME.MUTED
    end)
end

local function stopFarm()
    FARM_ENABLED = false
    if farmThread then
        task.cancel(farmThread)
        farmThread = nil
    end
    clearAllChams()
    status.Text = "Status: Off"
    status.TextColor3 = THEME.MUTED
end

toggleBtn.MouseButton1Click:Connect(function()

    setButtonState(toggleBtn, FARM_ENABLED)
    
    if FARM_ENABLED then
        IgnoredObjects = {} 
        startFarm()
    else
        stopFarm()
    end
end)
